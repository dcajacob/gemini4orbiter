cmake_minimum_required(VERSION 3.24)
project(Gemini4Orbiter LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
 message(FATAL_ERROR "This integration targets native Linux; use the original projects for legacy Windows")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ORBITER_NATIVE_SOURCE_DIR "" CACHE PATH "Matching native Orbiter source")
set(ORBITER_NATIVE_BUILD_DIR "" CACHE PATH "Matching native Orbiter build")
if(NOT EXISTS "${ORBITER_NATIVE_BUILD_DIR}/native-sdk/OrbiterNativeBuild.h")
 message(FATAL_ERROR "Set ORBITER_NATIVE_SOURCE_DIR and ORBITER_NATIVE_BUILD_DIR")
endif()
find_library(boundary orbiter_production_boundary PATHS "${ORBITER_NATIVE_BUILD_DIR}" NO_DEFAULT_PATH REQUIRED)
find_library(vessels orbiter_production_vessels PATHS "${ORBITER_NATIVE_BUILD_DIR}" NO_DEFAULT_PATH REQUIRED)
function(gemini_module name directory)
 add_library(${name} MODULE "Orbitersdk/Project Gemini/${directory}/${directory}.cpp" Native/Module.cpp)
 target_include_directories(${name} PRIVATE Native Orbitersdk/include)
 target_include_directories(${name} SYSTEM PRIVATE "${ORBITER_NATIVE_SOURCE_DIR}/Orbitersdk/include" "${ORBITER_NATIVE_SOURCE_DIR}/Src/Orbiter" "${ORBITER_NATIVE_BUILD_DIR}/native-sdk")
 target_compile_definitions(${name} PRIVATE GEMINI_NATIVE_ORBITER GEMINI_MODULE_NAME="${name}" ORBITER_NATIVE_BUILD_CONFIGURATION="$<CONFIG>")
 target_compile_options(${name} PRIVATE -Wno-write-strings)
 target_link_libraries(${name} PRIVATE "${vessels}" "${boundary}")
 target_link_options(${name} PRIVATE -Wl,--no-undefined)
 set_target_properties(${name} PROPERTIES PREFIX "" CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/NativeModules/Vessel")
 install(TARGETS ${name} LIBRARY DESTINATION NativeModules/Vessel)
endfunction()
gemini_module(Gemini_Titan Gemini_Titan)
gemini_module(GATV GATV)
gemini_module(gemini_astro Gemini_Astro)
gemini_module(gemini_eva gemini_eva)
gemini_module(Beacon Beacon)
target_sources(Gemini_Titan PRIVATE Native/Bitmaps.cpp)
